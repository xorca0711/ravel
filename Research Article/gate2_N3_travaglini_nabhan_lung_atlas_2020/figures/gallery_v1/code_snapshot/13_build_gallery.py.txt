def main():
    out=new_run(PACKAGE/'figures/gallery_v1');writer=PdfWriter();inputs={}
    lines=['# Nb4: research figure gallery','',
           'Twelve figures from the executed source reconstruction, sensitivities, external pilot and genome-wide visual analysis. White backgrounds, consistent assay colors, panel letters, donor points, explicit units and 300-dpi PNG plus SVG/PDF exports follow the neighboring Nb1, England and Nb3 galleries. Dense cell layers are rasterized within vector figures.','',
           '[12-page PDF atlas](figures/gallery_v1/Nb4_figure_atlas.pdf) · [Reproduction](reports/REPRODUCTION_REVIEW.md) · [Follow-up](reports/FOLLOWUP_RESULTS.md) · [Genome-wide analysis](reports/EXTENDED_VISUAL_ANALYSIS.md) · [RQ derivation](reports/RQ_DERIVATION.md)','',
           'These are descriptive and exploratory figures. Donor counts, coverage holds, annotation exposure and gene-set nulls are stated in each caption; no repair mechanism or population-level significance follows from visual separation.','']
    for i,(stem,title,folder,data,caption) in enumerate(FIGURES,1):
        root=f'figures/{folder}/{stem}'
        pdf=PACKAGE/(root+'.pdf');assert len(PdfReader(pdf).pages)==1
        writer.append(pdf);inputs[pdf.relative_to(PACKAGE).as_posix()]=sha256(pdf)
        lines += [f'## Figure {i}. {title}','',f'![{title}]({root}.png)','',f'[PNG]({root}.png) · [SVG]({root}.svg) · [PDF]({root}.pdf) · [Data]({data})','',caption,'']
    writer.write(out/'Nb4_figure_atlas.pdf')
    lines += ['## Render and data provenance','',
              'Renderers: [Figures 1–5](scripts/09_make_figures.py), [Figures 6–12](scripts/12_extended_figures.py), [gallery builder](scripts/13_build_gallery.py). Each finalized figure directory has an input manifest, output hashes and code snapshot. [Source renders](figures/publication_v1/run_record.json), [extended renders](figures/extended_v2/run_record.json), [atlas assembly](figures/gallery_v1/run_record.json).','',
              'The initial execution_v1 figures are retained as historical drafts. extended_v1 was an incomplete rendering attempt that stopped during a diagnostic for a size-filtered gene set; extended_v2 is the finalized version. Numerical analyses were not rerun for that rendering correction.','']
    (PACKAGE/'FIGURES.md').write_text('\n'.join(lines),encoding='utf-8')
    (PACKAGE/'figures/README.md').write_text('# Nb4 figures\n\nOpen the article-level [figure gallery](../FIGURES.md) for all 12 figures, captions, source data and PDF atlas.\n\nFinal exports: publication_v1 (Figures 1–5), extended_v2 (Figures 6–12), gallery_v1 (combined atlas). execution_v1 and incomplete extended_v1 are historical render attempts.\n',encoding='utf-8')
    write_json(out/'atlas_inputs.json',inputs)
    finish_record(out,{'schema':'Nb4-figure-atlas/v1','completed_at_utc':datetime.now(timezone.utc).isoformat(),'pages':12,'status':'article-local gallery and atlas assembled'})
    print('Wrote FIGURES.md and the 12-page Nb4 PDF figure atlas.')


if __name__=='__main__':main()
